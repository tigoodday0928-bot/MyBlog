{{/* 提示框 Callout Shortcode（可切換 tip / warning / info 三種預設樣式） -------------------------------- 用法： {{< callout type="tip" title="值得注意的是" >}} 張繼寫這首詩的時代背景是動亂流亡... {{< /callout >}} {{< callout type="warning" title="重要提醒" >}} - 中國高鐵採實名制... - 蘇州的高鐵站有好幾個... - 中國高鐵可以免費改簽一次... {{< /callout >}} 參數： type 選填，預設 tip。可用值：tip（💡 提示）／warning（⚠️ 警示）／info（ℹ️ 資訊） title 選填，不填則不顯示標題那一行 說明： 內容支援完整 markdown 語法（清單、粗體、連結等）。 之後若要再加新的類型（例如 danger、success）， 只要在下面的 $icons 字典跟最下面的顏色區塊各加一組即可。 */}} {{ $type := .Get "type" | default "tip" }} {{ $icons := dict "tip" "💡" "warning" "⚠️" "info" "ℹ️" }} {{ $icon := index $icons $type | default "💡" }}
{{ with .Get "title" }}
{{ $icon }} {{ . }}
{{ end }}
